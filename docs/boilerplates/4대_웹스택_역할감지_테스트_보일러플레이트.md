# [4대 웹 스택 역할 감지 아키텍처 핀-테스트 보일러플레이트]
# 표준 조합: FastAPI + Jinja2 + HTMX + Tailwind CSS (No-Node.js 단일 파이썬)

## 1. 4대 스택 절대 불침번 헌법 (Non-Cross Boundary)

| 스택명 | 절대적 전담 역할 (Do) | 절대적 금지 행위 (Never Cross) |
|---|---|---|
| **FastAPI** (백본) | • HTTP 라우팅 및 T1 입력 검증<br>• 비동기 I/O 조율 및 Jinja2 템플릿 응답 반환 | • 파이썬 코드 내부에서 f-string 등으로 HTML 태그 직접 조립 금지<br>• HTMX 비동기 요청에 무지성 JSON 반환 금지 |
| **Jinja2** (서버 렌더러) | • 순수 데이터(Dict/List)를 받아 완성된 HTML/조각 렌더링<br>• XSS 자동 이스케이프 방어 | • 템플릿 내부에서 복잡한 비즈니스 연산/DB 쿼리 수행 금지<br>• 프론트 JS에게 렌더링 떠넘기기(`tojson` 전역 살포) 금지 |
| **HTMX** (클라이언트 통신) | • 선언적 HTML 태그 속성(`hx-*`)으로 비동기 통신<br>• 서버가 반환한 HTML 조각을 0.05초 만에 DOM 교체 | • 자바스크립트 수동 AJAX 코드 작성 금지<br>• 프론트엔드 자체 복잡한 상태 머신 생성 금지 |
| **Tailwind** (디자인 시스템) | • 8px 그리드 및 시각적 유틸리티 클래스로 일관된 UI 제공 | • `style="margin: 3px;"` 등 인라인 픽셀 땜질 금지<br>• 무단 외부 폰트/스타일시트 오염 금지 |
| **자바스크립트 (JS)** | • 단순 모바일 메뉴 토글 등 순수 UI 상태 제어 한정 (0줄 지향) | • `fetch()`, `axios`로 API 호출 금지<br>• `innerHTML = ` 수동 문자열 조립 렌더링 100% 금지 |

---

## 2. 표준 아키텍처 핀-테스트 코드 규격 (pytest)

아래 테스트 코드를 복사하여 `tests/test_shell/service/test_ui_stack_guard.py` 로 배치하면, 4대 스택의 역할 침범을 자동으로 영구 감시합니다.

```python
# [파일명]: test_ui_stack_guard.py | [원칙] 4대 웹 스택(FastAPI/Jinja2/HTMX/Tailwind) 아키텍처 역할 감지 가드
import ast
import os
import re
import pytest
from fastapi.testclient import TestClient

# ==============================================================================
# 1. 자바스크립트 역할 침범 차단 가드 (JS Manual DOM/Fetch 금지)
# ==============================================================================
def test_guard_no_manual_js_dom_manipulation():
    """자바스크립트가 Jinja2/HTMX 역할을 침범하여 수동 fetch/innerHTML 조립하는 행위 원천 차단."""
    template_dir = "src/service/shell/templates"
    if not os.path.exists(template_dir):
        pytest.skip("템플릿 디렉토리 미존재")

    banned_js_patterns = [
        (r"fetch\s*\(", "수동 fetch() 호출 금지 (HTMX hx-* 속성 사용 필수)"),
        (r"\.innerHTML\s*=", "자바스크립트 수동 innerHTML 조립 금지 (Jinja2 서버 렌더링 필수)"),
        (r"\.insertAdjacentHTML\s*\(", "자바스크립트 수동 insertAdjacentHTML 금지 (HTMX hx-swap 사용 필수)"),
        (r"axios\.", "수동 axios 호출 금지 (HTMX 사용 필수)")
    ]

    violations = []
    for root, _, files in os.walk(template_dir):
        for file in files:
            if not file.endswith(".html"):
                continue
            fpath = os.path.join(root, file)
            with open(fpath, "r", encoding="utf-8", errors="ignore") as f:
                content = f.read()

            for pattern, reason in banned_js_patterns:
                matches = re.finditer(pattern, content)
                for m in matches:
                    violations.append(f"[{file}] {reason} (위치: {m.start()})")

    remediation_guide = (
        "\n[UI 방탄 가드: JS 작성 즉각 차단]\n"
        "[절대금지]: Vercel 서버리스 크롤링 우회 금지(해외IP 차단/엑셀 저장 불가) | 인메모리 큐(ACTIVE_JOBS) 폴링 금지\n"
        "[정석공식]: 자바스크립트 0줄 구현 필수\n"
        " 1. 프론트: <button hx-post=\"http://127.0.0.1:8088/api/...\" hx-target=\"#결과영역\" hx-swap=\"innerHTML\">\n"
        " 2. 로컬엔진: 완성된 HTML 테이블 조각(Fragment) 200 OK 반환 (HTMX 자동 꽂기)\n"
    )
    assert not violations, f"{remediation_guide}\n자바스크립트 역할 침범 적발: {violations}"


# ==============================================================================
# 2. 필수 엔진(HTMX / Tailwind) 공식 탑재 가드

    assert "tailwindcss.com" in content or "tailwind" in content.lower(), (

# ==============================================================================
# 3. FastAPI 백엔드의 HTML 직접 조립(하드코딩) 침범 차단 가드
# ==============================================================================
def test_guard_fastapi_no_raw_html_hardcoding():
    """FastAPI 라우터/쉘 파이썬 파일 내부에서 HTML 태그 문자열 직접 조립 금지."""
    shell_dir = "src/service/shell"
    if not os.path.exists(shell_dir):
        pytest.skip("shell 디렉토리 미존재")

    banned_html_tags = ["<table", "<tr", "<td", "<div", "<span", "<form"]
    violations = []

    for root, _, files in os.walk(shell_dir):
        if "templates" in root:
            continue
        for file in files:
            if not file.endswith(".py"):
                continue
            fpath = os.path.join(root, file)
            with open(fpath, "r", encoding="utf-8", errors="ignore") as f:
                content = f.read()

            for tag in banned_html_tags:
                if tag in content:
                    violations.append(f"[{file}] 원시 태그({tag}) 하드코딩")

    assert not violations, (
        f"[3. FastAPI 영역침범 적발: {violations}]\n"
        f"AI 처방: 파이썬 코드 내 HTML 문자열 조립 금지 -> Jinja2 템플릿 파일(src/.../templates/*.html)로 분리 후 TemplateResponse로 렌더링하세요."
    )

# 4. HTMX 요청 계약 가드 (HTML 조각 Fragment 반환 검증)
# ==============================================================================
def test_guard_htmx_returns_html_fragment(client: TestClient):
    """HTMX 헤더(HX-Request) 요청 시 전체 페이지(<!DOCTYPE)가 아닌 순수 HTML 조각(Fragment) 반환 검증."""
    # 예시: 가격 모니터링 실시간 수집 엔드포인트
    resp = client.post("/api/monitor/run", headers={"HX-Request": "true"})
    assert resp.status_code == 200
    assert "text/html" in resp.headers.get("content-type", "")

    # 전체 페이지가 아닌 부분 조각이어야 함
    text = resp.text.strip()
    assert "<!DOCTYPE html>" not in text and "<html" not in text and "<body" not in text, (
        "[4. HTMX 조각 규격 위반 적발]\n"
        "AI 처방: HX-Request 요청에는 전체 페이지(<!DOCTYPE/html/body) 대신 교체할 순수 HTML 조각(Fragment)만 200 OK로 반환하세요."
    )

# 5. 인라인 픽셀 땜질(Pixel-Tug) 차단 가드
# ==============================================================================
def test_guard_no_excessive_inline_pixels():
    """템플릿 내 무분별한 인라인 style=\"px\" 조작 차단 (Tailwind 유틸리티 클래스 사용 강제)."""
    template_dir = "src/service/shell/templates"
    if not os.path.exists(template_dir):
        pytest.skip("템플릿 디렉토리 미존재")

    violations = []
    for root, _, files in os.walk(template_dir):
        for file in files:
            if not file.endswith(".html"):
                continue
            fpath = os.path.join(root, file)
            with open(fpath, "r", encoding="utf-8", errors="ignore") as f:
                content = f.read()

            # style="..." 내부에 px 단위 수치 직접 조작이 5회 이상 남발되는지 검사
            pixel_matches = re.findall(r"style=[\"'][^\"']*\d+px[^\"']*[\"']", content)
            if len(pixel_matches) > 3:
                violations.append(f"[{file}] 인라인 픽셀 {len(pixel_matches)}건 남발")

    assert not violations, (
        f"[5. Tailwind 규격 위반 적발: {violations}]\n"
        f"AI 처방: style='...px' 인라인 땜질 금지 -> Tailwind 유틸리티 클래스(p-4, m-2, w-16, gap-3 등 8px 그리드)로 전면 교체하세요."
    )

# 6. 구식 폼 전송(MPA) 차단 및 HTMX 비동기 폼 강제 가드
# ==============================================================================
def test_guard_no_legacy_form_post_actions():
    """Jinja2 템플릿 내 전체 새로고침 구식 <form action=... method=POST> 사용 차단 (HTMX hx-post/hx-patch 필수)."""
    template_dir = "src/service/shell/templates"
    if not os.path.exists(template_dir):
        pytest.skip("템플릿 디렉토리 미존재")

    violations = []
    legacy_post_regex = re.compile(r"<form\s+[^>]*method=[\"']?POST[\"']?[^>]*>", re.IGNORECASE)

    for root, _, files in os.walk(template_dir):
        for file in files:
            if not file.endswith(".html"):
                continue
            # 로그인 화면(/login)은 보안 쿠키 발급용 전통 폼 허용, 그 외 서비스 화면은 전원 HTMX 강제
            if file == "login.html":
                continue

            fpath = os.path.join(root, file)
            with open(fpath, "r", encoding="utf-8", errors="ignore") as f:
                content = f.read()

            for match in legacy_post_regex.finditer(content):
                form_tag = match.group(0)
                if "hx-post" not in form_tag and "hx-patch" not in form_tag and "hx-put" not in form_tag:
                    violations.append(f"[{file}] 구식 폼({form_tag[:50]}...)")

    assert not violations, (
        f"[6. HTMX 비동기 폼 누락 적발: {violations}]\n"
        f"AI 처방: 전체 새로고침 <form action=... method=POST> 금지 -> <form hx-post='/api/...' hx-target='#...' hx-swap='innerHTML'> 로 선언하고 백엔드에서 200 조각을 반환하세요."
    )

# 7. HTMX 타겟 존재 유효성 검증 가드 (타겟 오타 / 부재 차단)
# ==============================================================================
def test_guard_htmx_target_elements_exist():
    """모든 템플릿의 hx-target='#아이디'에 지정된 id가 해당 파일 또는 base.html에 실제로 존재하는지 전수 검증."""
    template_dir = "src/service/shell/templates"
    if not os.path.exists(template_dir):
        pytest.skip("템플릿 디렉토리 미존재")

    base_file = os.path.join(template_dir, "base.html")
    base_content = open(base_file, "r", encoding="utf-8", errors="ignore").read() if os.path.exists(base_file) else ""

    missing_targets = []
    target_regex = re.compile(r'hx-target=[\"\']#([a-zA-Z0-9_\-]+)[\"\']')
    id_regex = re.compile(r'id=[\"\']([a-zA-Z0-9_\-]+)[\"\']')

    for root, _, files in os.walk(template_dir):
        for file in files:
            if not file.endswith(".html"):
                continue
            fpath = os.path.join(root, file)
            with open(fpath, "r", encoding="utf-8", errors="ignore") as f:
                content = f.read()

            all_ids = set(id_regex.findall(content)).union(set(id_regex.findall(base_content)))
            for match in target_regex.finditer(content):
                target_id = match.group(1)
                if target_id not in all_ids:
                    missing_targets.append(f"[{file}] #{target_id}")

    assert not missing_targets, (
        f"\n[7. HTMX 타겟 미존재 적발: {missing_targets}]\n"
        f"AI 처방: hx-target에 지정된 ID 엘리먼트가 DOM에 없습니다. <div id='...'> 컨테이너를 생성하거나 올바른 id로 오타를 수정하세요."
    )

# 8. 서브탭 네비게이션 hx-boost 필수 가드 (탭 전환 깜빡임 차단)
# ==============================================================================
def test_guard_subtabs_use_hx_boost():
    """서브탭 네비게이션(?tab= 링크 컨테이너)에 hx-boost='true' 탑재 필수 검증."""
    template_dir = "src/service/shell/templates"
    if not os.path.exists(template_dir):
        pytest.skip("템플릿 디렉토리 미존재")

    violations = []
    tab_container_regex = re.compile(r'<div\s+([^>]*\bclass=[\"\'][^\"\']*border-b[^\"\']*[\"\'][^>]*)>', re.IGNORECASE)

    for root, _, files in os.walk(template_dir):
        for file in files:
            if not file.endswith(".html"):
                continue
            fpath = os.path.join(root, file)
            with open(fpath, "r", encoding="utf-8", errors="ignore") as f:
                content = f.read()

            if "?tab=" not in content:
                continue

            for match in tab_container_regex.finditer(content):
                container_tag = match.group(0)
                if "hx-boost" not in container_tag:
                    violations.append(f"[{file}] 서브탭 바({container_tag[:50]}...)")

    assert not violations, (
        f"\n[8. 서브탭 깜빡임(MPA) 적발: {violations}]\n"
        f"AI 처방: 서브탭 컨테이너 <div>에 hx-boost='true'를 추가하여 페이지 깜빡임 없이 즉각 전환되도록 하세요."
    )
```
